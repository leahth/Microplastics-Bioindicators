Toxicity Scoring

This file assigns toxicity scores to species in the ToMEx database for the prioritization of candidate bioindicators for microplastics. Each species is scored based on 4 criteria within the toxicity category - sensitivity, dose response, level of biological organization, and weight of evidence. # Setup

## Selecting concentration in each matrix as "Environmentally relevant" threshold
envrconc_SW <- max(corrected_concs_water$corrected_particle_per_mL)
envrconc_SW_plusOM <- envrconc_SW*10

#for each species, take the lowest dose at which a significant effect was observed, then score against the environmental concentration thresholds
species_sensitivity <- marinewater_particle_aligned |>
    group_by(Species) |>
    summarize(`Organism Group` = first(`Organism Group`), min_dose = min(`Plotted Dose Values`, na.rm = TRUE)) |>
    mutate(Sensitivity = case_when(
      min_dose < envrconc_SW ~ 2,
      min_dose < envrconc_SW_plusOM ~ 1,
      TRUE ~ 0
    )) |>
    select(Species, `Organism Group`, Sensitivity) |>
    arrange(desc(Sensitivity))

Dose Response

0 – No data sets available with at least 3 test concentrations in series or data sets with 3+ test concentrations do not demonstrate that response increases with increasing dose 1 – At least one data set available with 3-4 test concentrations, response increases with increasing dose 2 – At least one data set available with 5 test concentrations, response increases with increasing dose

Code 
#Currently using "Effect == YES" as a proxy for increasing dose. Need to individually evaluate magnitude.
#`Number of Doses` is in addition to the control

#Magnitude of response isn't available in this data, so as a proxy: a data set "demonstrates" dose-response if, ordered by dose,
#Effect never reverts from "Yes" back to "No" at a higher dose (once an effect appears, it persists at higher doses).
species_dose_response <- marinewater_particle_aligned_import |>
    group_by(Species, DOI, Endpoint, `Broad Endpoint Category`, `Specific Endpoint Category`, `Exposure Route`, Polymer, Shape, `Size Category`, Sex, `Exposure Duration (days)`, `Number of Doses`, `Life Stage`, `In vitro/in vivo`) |>
    arrange(`Plotted Dose Values`, .by_group = TRUE) |>
    summarise(
      monotonic = {
        has_effect <- any(Effect == "Yes")
        if (!has_effect) {
          #effect never appears at any dose in this series, so no dose-response relationship to speak of
          FALSE
        } else {
          #index of the lowest dose where an effect first shows up (rows are already sorted by dose)
          first_yes_position <- min(which(Effect == "Yes"))
          #effect status at that dose and every higher dose after it
          effect_from_first_yes_onward <- Effect[first_yes_position:n()]
          #TRUE only if the effect stays "Yes" all the way up; FALSE if it ever reverts back to "No"
          all(effect_from_first_yes_onward == "Yes")
        }
      },
      .groups = "drop"
    ) |>
    mutate(dataset_score = case_when(
      monotonic & `Number of Doses` >= 5 ~ 2,
      monotonic & `Number of Doses` >= 3 ~ 1,
      TRUE ~ 0
    )) |>
    group_by(Species) |>
    summarize(`Dose Response` = max(dataset_score, na.rm = TRUE))

species_sensitivity <- species_sensitivity |>
    left_join(species_dose_response, by = "Species") |>
    arrange(desc(Sensitivity), desc(`Dose Response`))

Level of Biological Organization

0 – Effects only observed below the organism level 1 – Effects observed at the organism and population levels 2 – Effects observed at the organism and population levels pertaining to growth, development, reproduction, and survival

Code 
#Score 2 requires an effect at the organism/population level AND that effect pertaining to growth, development, reproduction, or survival (mortality)
species_bio_org <- marinewater_particle_aligned |>
    group_by(Species) |>
    summarize(
      has_organism_pop_effect = any(`Level of Biological Organization` %in% c("Organism", "Population")),
      has_gdrs_effect = any(`Level of Biological Organization` %in% c("Organism", "Population") &
                                          `Specific Endpoint Category` %in% c("Mortality", "Growth", "Reproduction", "Development"))
    ) |>
    mutate(`Level of Biological Organization` = case_when(
      has_gdrs_effect ~ 2,
      has_organism_pop_effect ~ 1,
      TRUE ~ 0
    )) |>
    select(Species, `Level of Biological Organization`)

species_sensitivity <- species_sensitivity |>
    left_join(species_bio_org, by = "Species") |>
    arrange(desc(Sensitivity), desc(`Dose Response`))

Weight of Evidence

0 – Only 1 available study available or 2 studies with directly conflicting results 1 – At least 2 available studies and results generally agree 2 – More than 2 available studies and results generally agree

Summary toxicity scores

    arrange(desc(`Total Score`))
